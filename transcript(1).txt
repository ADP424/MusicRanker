 * Build a map from genre id → set of lowercased ancestor names.
 * Walking parent_ids recursively so that searching "Action" also
 * surfaces children like "Action Comedy" or "Action Thriller".
 */
function buildAncestorMap(genres: MovieGenre[]): Map<number, Set<string>> {
  const byId = new Map(genres.map((g) => [g.id, g]));
  const cache = new Map<number, Set<string>>();

  function resolve(id: number, visited: Set<number>): Set<string> {
    if (cache.has(id)) return cache.get(id)!;
    const result = new Set<string>();
    const genre = byId.get(id);
    if (genre) {
      for (const pid of genre.parent_ids) {
        if (visited.has(pid)) continue;
        const parent = byId.get(pid);
        if (parent) {
          result.add(parent.name.toLowerCase());
          visited.add(pid);
          for (const a of resolve(pid, visited)) result.add(a);
        }
      }
    }
    cache.set(id, result);
    return result;
  }

  for (const g of genres) resolve(g.id, new Set([g.id]));
  return cache;
}

/**
 * Filter movie genres by a search string, matching on:

 *   - genre synonyms
